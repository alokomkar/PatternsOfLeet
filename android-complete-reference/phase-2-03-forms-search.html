Filter function

fun filterBusinesses(
    businesses: List<Business>,
    query: String,
    category: String?
): List<Business> {
    val normalizedQuery = query.trim().lowercase()

    return businesses.filter { business ->
        val matchesQuery =
            business.name.lowercase().contains(normalizedQuery) ||
            business.city.lowercase().contains(normalizedQuery)

        val matchesCategory =
            category == null || business.category == category

        matchesQuery && matchesCategory
    }
}
